<script>
  const marques = [
    { nom:"regab", grande:600, petite:350 },
    { nom:"33 export", grande:650, petite:400 },
    { nom:"castel", grande:600, petite:350 },
    { nom:"booster", grande:1500, petite:1000 },
    { nom:"guinness", grande:1200, petite:700 },
    { nom:"beaufort", grande:700, petite:450 },
    { nom:"tembo", grande:600, petite:350 },
    { nom:"heineken", grande:1000, petite:600 },
    { nom:"malta guinness", grande:900, petite:500 },
    { nom:"coca-cola", grande:600, petite:350 },
  ];

  const grid = document.getElementById("brandGrid");
  let selection = null; // { marque, taille, prix, quantite }

  marques.forEach((m, idx) => {
    const cell = document.createElement("div");
    cell.className = "brand-cell";
    cell.dataset.idx = idx;
    cell.innerHTML = `
      <div class="nom">${m.nom}</div>
      <div class="tailles">
        <div class="taille-chip" data-taille="grande">grande · ${m.grande}</div>
        <div class="taille-chip" data-taille="petite">petite · ${m.petite}</div>
      </div>
    `;
    grid.appendChild(cell);
  });

  grid.addEventListener("click", (e) => {
    const chip = e.target.closest(".taille-chip");
    if (!chip) return;
    const cellEl = chip.closest(".brand-cell");
    const idx = +cellEl.dataset.idx;
    const taille = chip.dataset.taille;
    const m = marques[idx];
    const prix = taille === "grande" ? m.grande : m.petite;

    document.querySelectorAll(".brand-cell").forEach(c => c.classList.remove("selected"));
    document.querySelectorAll(".taille-chip").forEach(c => c.classList.remove("active"));
    cellEl.classList.add("selected");
    chip.classList.add("active");

    selection = { marque: m.nom, taille, prix, quantite: 1 };
    updateDisplay();
  });

  // Cellule "autres" — se déplie vers le bas
  const autresCell = document.getElementById("autresCell");
  const autresListe = document.getElementById("autresListe");
  autresCell.addEventListener("click", () => {
    autresCell.classList.toggle("open");
    autresListe.classList.toggle("open");
  });
  autresListe.addEventListener("click", (e) => {
    const item = e.target.closest(".autres-item");
    if (!item) return;
    document.querySelectorAll(".brand-cell").forEach(c => c.classList.remove("selected"));
    document.querySelectorAll(".taille-chip").forEach(c => c.classList.remove("active"));
    selection = { marque: item.dataset.nom, taille: null, prix: +item.dataset.prix, quantite: 1 };
    updateDisplay();
    autresCell.classList.remove("open");
    autresListe.classList.remove("open");
  });

  // Clavier — construit la quantité
  let saisie = "";
  document.querySelectorAll(".keypad .key").forEach(key => {
    key.addEventListener("click", () => {
      const txt = key.textContent.trim();
      if (key.classList.contains("effa")) { saisie = ""; }
      else if (key.classList.contains("del")) { saisie = saisie.slice(0, -1); }
      else if (key.classList.contains("eq")) { /* pas de calcul libre dans la maquette */ }
      else if (!key.classList.contains("op")) { saisie += txt; }
      if (selection) {
        selection.quantite = saisie === "" ? 1 : parseInt(saisie, 10) || 1;
      }
      updateDisplay();
    });
  });

  document.getElementById("btnAbc").addEventListener("click", () => {
    document.getElementById("dispSelection").textContent = "recherche produit (texte) — à venir";
  });

  function updateDisplay() {
    const dispSelection = document.getElementById("dispSelection");
    const dispFormule = document.getElementById("dispFormule");
    const dispTotal = document.getElementById("dispTotal");

    if (!selection) {
      dispSelection.textContent = "aucune sélection";
      dispFormule.innerHTML = "&nbsp;";
      dispTotal.innerHTML = '0 <span class="devise">fcfa</span>';
      return;
    }
    const libelle = selection.taille ? `${selection.marque} (${selection.taille})` : selection.marque;
    dispSelection.textContent = libelle;
    dispFormule.textContent = `${selection.prix} × ${selection.quantite}`;
    const total = selection.prix * selection.quantite;
    dispTotal.innerHTML = `${total.toLocaleString("fr-FR")} <span class="devise">fcfa</span>`;
  }
</script>
